CREATE TABLE IF NOT EXISTS transactions(id UUID DEFAULT gen_random_uuid()PRIMARY KEY,user_name TEXT,type TEXT CHECK(type IN('income','expense')),category TEXT,amount NUMERIC(12,2),description TEXT DEFAULT'',date DATE DEFAULT now(),cuota_info TEXT DEFAULT'',card_name TEXT DEFAULT'',created_at TIMESTAMPTZ DEFAULT now());
CREATE TABLE IF NOT EXISTS accounts(id UUID DEFAULT gen_random_uuid()PRIMARY KEY,name TEXT,type TEXT DEFAULT'bank',balance NUMERIC(12,2)DEFAULT 0,color TEXT DEFAULT'#E8A838',icon TEXT DEFAULT'fa-building-columns',card_brand TEXT DEFAULT'',created_at TIMESTAMPTZ DEFAULT now());
CREATE TABLE IF NOT EXISTS goals(id UUID DEFAULT gen_random_uuid()PRIMARY KEY,name TEXT,target NUMERIC(12,2),current NUMERIC(12,2)DEFAULT 0,color TEXT DEFAULT'#E8A838',deadline DATE,created_at TIMESTAMPTZ DEFAULT now());
CREATE TABLE IF NOT EXISTS installments(id UUID DEFAULT gen_random_uuid()PRIMARY KEY,description TEXT,merchant TEXT DEFAULT'',total_amount NUMERIC(12,2),cuota_amount NUMERIC(12,2),total_cuotas INTEGER,paid_cuotas INTEGER DEFAULT 0,category TEXT DEFAULT'otros',user_name TEXT,card_name TEXT DEFAULT'',start_date DATE,created_at TIMESTAMPTZ DEFAULT now());
ALTER TABLE transactions ENABLE ROW LEVEL SECURITY;ALTER TABLE accounts ENABLE ROW LEVEL SECURITY;ALTER TABLE goals ENABLE ROW LEVEL SECURITY;ALTER TABLE installments ENABLE ROW LEVEL SECURITY;
CREATE POLICY"p1"ON transactions FOR ALL USING(true)WITH CHECK(true);CREATE POLICY"p2"ON accounts FOR ALL USING(true)WITH CHECK(true);CREATE POLICY"p3"ON goals FOR ALL USING(true)WITH CHECK(true);CREATE POLICY"p4"ON installments FOR ALL USING(true)WITH CHECK(true);

CREATE TABLE transactions (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  user_name TEXT NOT NULL,
  type TEXT NOT NULL CHECK (type IN ('income','expense')),
  category TEXT NOT NULL,
  amount NUMERIC(12,2) NOT NULL,
  description TEXT DEFAULT '',
  date DATE NOT NULL DEFAULT now(),
  account_id UUID,
  cuota_info TEXT DEFAULT '',
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE accounts (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  type TEXT NOT NULL DEFAULT 'bank',
  balance NUMERIC(12,2) NOT NULL DEFAULT 0,
  color TEXT DEFAULT '#E8A838',
  icon TEXT DEFAULT 'fa-building-columns',
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE goals (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  name TEXT NOT NULL,
  target NUMERIC(12,2) NOT NULL,
  current NUMERIC(12,2) NOT NULL DEFAULT 0,
  icon TEXT DEFAULT 'fa-bullseye',
  color TEXT DEFAULT '#E8A838',
  deadline DATE,
  created_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE installments (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  description TEXT NOT NULL,
  merchant TEXT DEFAULT '',
  total_amount NUMERIC(12,2) NOT NULL,
  cuota_amount NUMERIC(12,2) NOT NULL,
  total_cuotas INTEGER NOT NULL,
  paid_cuotas INTEGER NOT NULL DEFAULT 0,
  category TEXT DEFAULT 'otros',
  user_name TEXT NOT NULL,
  card_name TEXT DEFAULT '',
  start_date DATE,
  created_at TIMESTAMPTZ DEFAULT now()
);

ALTER TABLE transactions ENABLE ROW LEVEL SECURITY;

CREATE POLICY "g_all" ON goals FOR ALL USING (true) WITH CHECK (true);
ALTER TABLE installments ENABLE ROW LEVEL SECURITY;
CREATE POLICY "i_all" ON installments FOR ALL USING (true) WITH CHECK (true);
